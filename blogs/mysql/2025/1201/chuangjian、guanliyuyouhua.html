MySQL索引完全指南：创建、管理与优化

前言

索引是数据库性能优化的核心。正确的索引策略可以让查询速度提升几个数量级，而不当的索引则会影响写入性能并占用额外存储空间。本文将全面介绍MySQL索引的创建、管理与维护。

一、索引创建语法大全

1.1 基础索引创建

1. 创建普通索引

-- 单列索引
CREATE INDEX idx_name ON employees(last_name);

-- 多列复合索引
CREATE INDEX idx_name_dept ON employees(last_name, department_id);

2. 创建唯一索引

-- 保证列值的唯一性
CREATE UNIQUE INDEX idx_email ON users(email);

-- 多列组合唯一索引
CREATE UNIQUE INDEX idx_user_product ON orders(user_id, product_id);

3. 创建主键索引

-- 创建表时指定主键
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100)
);

-- 为已存在表添加主键
ALTER TABLE employees ADD PRIMARY KEY (employee_id);

1.2 高级索引选项

1. 前缀索引（字符串索引优化）

-- 只对字符串的前N个字符创建索引
CREATE INDEX idx_name_prefix ON customers(name(20));

2. 全文索引（支持全文搜索）

-- 对文章内容创建全文索引
CREATE FULLTEXT INDEX idx_content ON articles(content);

-- 使用全文搜索
SELECT * FROM articles 
WHERE MATCH(content) AGAINST('数据库 优化' IN NATURAL LANGUAGE MODE);

3. 空间索引（地理位置数据）

CREATE SPATIAL INDEX idx_location ON places(coordinates);

4. 函数索引（MySQL 8.0+）

-- 基于表达式创建索引
CREATE INDEX idx_lower_name ON users((LOWER(name)));

-- 使用函数索引查询
SELECT * FROM users WHERE LOWER(name) = 'john';

1.3 创建表时定义索引

CREATE TABLE employees (
    id INT AUTO_INCREMENT,
    first_name VARCHAR(50),
    last_name VARCHAR(50),
    email VARCHAR(100),
    hire_date DATE,
    salary DECIMAL(10,2),
    department_id INT,
    
    -- 主键索引
    PRIMARY KEY (id),
    
    -- 唯一索引
    UNIQUE KEY uk_email (email),
    
    -- 普通索引
    KEY idx_last_name (last_name),
    
    -- 复合索引
    KEY idx_dept_hire (department_id, hire_date),
    
    -- 前缀索引
    KEY idx_first_name_prefix (first_name(10)),
    
    -- 外键（自动创建索引）
    FOREIGN KEY (department_id) 
        REFERENCES departments(id)
        ON DELETE CASCADE
);

二、修改和删除索引

2.1 索引修改

-- MySQL中不能直接修改索引，需要先删除再创建
-- 1. 删除旧索引
DROP INDEX idx_name ON employees;

-- 2. 创建新索引
CREATE INDEX idx_new_name ON employees(last_name, first_name);

-- 或者使用ALTER TABLE语句
ALTER TABLE employees 
DROP INDEX idx_name,
ADD INDEX idx_new_name (last_name, first_name);

2.2 索引删除

-- 删除普通索引
DROP INDEX idx_name ON table_name;

-- 删除唯一索引
DROP INDEX uk_email ON users;

-- 删除主键索引
ALTER TABLE table_name DROP PRIMARY KEY;

-- 批量删除索引（不推荐生产环境使用）
SELECT CONCAT('DROP INDEX ', index_name, ' ON ', table_name, ';')
FROM information_schema.statistics
WHERE table_schema = 'your_database'
  AND table_name = 'your_table'
  AND index_name != 'PRIMARY';

三、查看索引信息

3.1 常用查看命令

-- 1. 显示表的所有索引
SHOW INDEX FROM employees;
SHOW INDEXES FROM employees;
SHOW KEYS FROM employees;

-- 2. 查看建表语句（包含索引信息）
SHOW CREATE TABLE employees;

-- 3. 从information_schema查看索引信息
SELECT 
    TABLE_NAME,
    INDEX_NAME,
    NON_UNIQUE,
    SEQ_IN_INDEX,
    COLUMN_NAME,
    CARDINALITY,
    INDEX_TYPE,
    COMMENT
FROM information_schema.STATISTICS
WHERE TABLE_SCHEMA = 'your_database'
  AND TABLE_NAME = 'employees'
ORDER BY INDEX_NAME, SEQ_IN_INDEX;

-- 4. 查看索引使用情况
SELECT * FROM sys.schema_index_statistics
WHERE table_schema = 'your_database';

-- 5. 查看未使用的索引（MySQL 8.0+）
SELECT * FROM sys.schema_unused_indexes;

3.2 索引分析工具

-- 1. 使用EXPLAIN分析索引使用情况
EXPLAIN SELECT * FROM employees 
WHERE last_name = 'Smith' AND department_id = 5;

-- 2. 查看索引大小
SELECT 
    TABLE_NAME,
    INDEX_NAME,
    ROUND(SUM(DATA_LENGTH + INDEX_LENGTH) / 1024 / 1024, 2) AS 'Size(MB)'
FROM information_schema.TABLES
WHERE TABLE_SCHEMA = 'your_database'
GROUP BY TABLE_NAME, INDEX_NAME;

-- 3. 查看索引碎片率
SELECT 
    table_name,
    index_name,
    round(data_free/(data_length+index_length)*100, 2) as碎片率
FROM information_schema.tables 
WHERE data_free > 0;

四、索引维护操作

4.1 索引重建与优化

-- 1. 重建索引（InnoDB）
ALTER TABLE table_name ENGINE=InnoDB;
-- 或
OPTIMIZE TABLE table_name;

-- 2. 在线重建索引（MySQL 5.6+）
ALTER TABLE table_name DROP INDEX idx_name, 
ALGORITHM=INPLACE, 
LOCK=NONE;

CREATE INDEX idx_name ON table_name(column_name),
ALGORITHM=INPLACE,
LOCK=NONE;

-- 3. 分析表（更新索引统计信息）
ANALYZE TABLE employees;

-- 4. 强制使用/忽略索引
SELECT * FROM employees USE INDEX (idx_name) WHERE ...;
SELECT * FROM employees IGNORE INDEX (idx_name) WHERE ...;
SELECT * FROM employees FORCE INDEX (idx_name) WHERE ...;

4.2 索引监控

-- 1. 开启索引监控（MySQL 8.0）
-- 在performance_schema中监控索引使用
UPDATE performance_schema.setup_instruments
SET ENABLED = 'YES', TIMED = 'YES'
WHERE NAME LIKE '%wait/io/table/sql/handler%';

-- 2. 查看索引使用统计
SELECT 
    OBJECT_SCHEMA,
    OBJECT_NAME,
    INDEX_NAME,
    COUNT_READ,
    COUNT_FETCH,
    COUNT_INSERT,
    COUNT_UPDATE,
    COUNT_DELETE
FROM performance_schema.table_io_waits_summary_by_index_usage
WHERE OBJECT_SCHEMA = 'your_database'
ORDER BY COUNT_READ DESC;

4.3 自动索引管理脚本

-- 自动删除重复索引
DELIMITER $$

CREATE PROCEDURE remove_duplicate_indexes()
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE v_table_schema VARCHAR(64);
    DECLARE v_table_name VARCHAR(64);
    DECLARE v_dup_indexes TEXT;
    
    DECLARE cur CURSOR FOR 
    SELECT table_schema, table_name, GROUP_CONCAT(index_name)
    FROM (
        SELECT table_schema, table_name, index_name,
               GROUP_CONCAT(column_name ORDER BY seq_in_index) AS columns
        FROM information_schema.statistics
        WHERE table_schema NOT IN ('mysql', 'information_schema', 'performance_schema')
        GROUP BY table_schema, table_name, index_name
    ) t
    GROUP BY table_schema, table_name, columns
    HAVING COUNT(*) > 1;
    
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
    
    OPEN cur;
    
    read_loop: LOOP
        FETCH cur INTO v_table_schema, v_table_name, v_dup_indexes;
        IF done THEN
            LEAVE read_loop;
        END IF;
        
        -- 保留第一个索引，删除其他重复索引
        SET @indexes = SUBSTRING_INDEX(v_dup_indexes, ',', -1);
        SET @sql = CONCAT('DROP INDEX ', @indexes, ' ON ', 
                         v_table_schema, '.', v_table_name);
        
        PREPARE stmt FROM @sql;
        EXECUTE stmt;
        DEALLOCATE PREPARE stmt;
        
        SELECT CONCAT('Removed duplicate index: ', @indexes, 
                     ' from ', v_table_schema, '.', v_table_name) AS message;
    END LOOP;
    
    CLOSE cur;
END$$

DELIMITER ;

-- 执行存储过程
CALL remove_duplicate_indexes();

五、最佳实践与常见问题

5.1 索引设计原则

  1. 选择性原则：优先为高选择性的列创建索引
  2. 最左前缀原则：复合索引要遵循查询条件
  3. 覆盖索引：尽量让查询只通过索引完成
  4. 索引数量控制：一般表不要超过5-7个索引

5.2 常见问题解决

-- 1. 索引失效场景排查
-- 使用函数导致索引失效
SELECT * FROM users WHERE UPPER(name) = 'JOHN'; -- 索引失效
SELECT * FROM users WHERE name = 'JOHN'; -- 使用索引

-- LIKE查询通配符开头导致索引失效
SELECT * FROM users WHERE name LIKE '%john%'; -- 索引失效
SELECT * FROM users WHERE name LIKE 'john%'; -- 使用索引

-- 2. 处理索引碎片
-- 定期优化表
SET GLOBAL innodb_defragment = 1;
OPTIMIZE TABLE large_table;

-- 3. 监控索引使用率
-- 查找从未使用的索引
SELECT * FROM sys.schema_unused_indexes
WHERE object_schema = 'your_database';

5.3 性能对比测试

-- 创建测试表
CREATE TABLE test_index (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100),
    email VARCHAR(100),
    age INT,
    created_at DATETIME
);

-- 插入100万测试数据
-- ...

-- 测试无索引查询
SELECT * FROM test_index WHERE email = 'test@example.com';
-- 执行时间: 约500ms

-- 创建索引后查询
CREATE INDEX idx_email ON test_index(email);
SELECT * FROM test_index WHERE email = 'test@example.com';
-- 执行时间: 约10ms (提升50倍)

六、总结

有效的索引管理需要：

  1. 定期审查：使用sys.schema_unused_indexes找出无用索引
  2. 监控性能：通过EXPLAIN和性能模式分析索引效果
  3. 适时优化：对碎片率高的索引进行重建
  4. 遵循最佳实践：避免过度索引和索引滥用
  5. 版本适配：根据不同MySQL版本使用新特性

记住：索引不是越多越好，合适的索引才是性能提升的关键。定期维护和优化索引策略，才能确保数据库长期高效运行。
